function drawTeamLog(doc, team, title, y){ const marginX = 40; const leftX = 40; const rightX = 310; const columnWidth = 230; const chip = 15; const chipGap = 4; const players = state.teams[team].players; doc.setFont('helvetica','bold'); doc.setFontSize(13); doc.setTextColor(...SLU_BLUE_RGB); doc.text(title + ' Passing — Attempt Log', marginX, y); doc.setTextColor(20); y += 18; if(players.length === 0){ doc.setFont('helvetica','normal'); doc.setFontSize(10); doc.text('No players recorded.', marginX, y); return y + 14; } let rowY = y; let leftBlockHeight = 0; for(let i = 0; i < players.length; i += 2){ const pair = [players[i], players[i + 1]].filter(Boolean); let tallestBlock = 0; pair.forEach((p, columnIndex) => { const startX = columnIndex === 0 ? leftX : rightX; if(rowY > 700){ doc.addPage(); rowY = 50; } doc.setFont('helvetica','bold'); doc.setFontSize(10.5); doc.setTextColor(20); doc.text(p.label, startX, rowY); doc.setFont('helvetica','normal'); doc.setFontSize(9); doc.setTextColor(110); doc.text( `${p.passes.length} attempt${p.passes.length === 1 ? '' : 's'}`, startX + 34, rowY ); doc.setTextColor(20); if(p.passes.length === 0){ tallestBlock = Math.max(tallestBlock, 34); doc.setFont('helvetica','normal'); doc.setFontSize(8); doc.setTextColor(130); doc.text('No passes recorded.', startX, rowY + 16); doc.setTextColor(20); return; } let chipX = startX; let chipY = rowY + 20; p.passes.forEach(pass => { if(chipX + chip > startX + columnWidth){ chipX = startX; chipY += chip + chipGap; } doc.setFillColor(...SET_RGB[pass.set]); doc.roundedRect( chipX, chipY - chip, chip, chip, 2.5, 2.5, 'F' ); doc.setFont('helvetica','bold'); doc.setFontSize(9); doc.setTextColor(255,255,255); doc.text( String(pass.value), chipX + chip / 2, chipY - chip / 2 + 3, { align:'center' } ); chipX += chip + chipGap; }); const blockHeight = (chipY - rowY) + 20; tallestBlock = Math.max(tallestBlock, blockHeight); }); rowY += tallestBlock + 12; } return rowY; }
